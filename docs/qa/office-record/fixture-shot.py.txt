import asyncio
from playwright.async_api import async_playwright
async def main():
  async with async_playwright() as p:
    b=await p.chromium.launch(headless=True)
    for w in (1280,375):
      pg=await b.new_page(viewport={"width":w,"height":1800}); await pg.goto("file:///tmp/fixture/page.html")
      h=await pg.evaluate("document.body.scrollHeight"); await pg.set_viewport_size({"width":w,"height":min(h+20,1800)})
      await pg.screenshot(path=f"/tmp/fixture/office-record-fixture-{w}.jpg", type="jpeg", quality=70)
    await b.close()
asyncio.run(main())
